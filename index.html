/* =========================================================
   NUMBER FORMATTING
========================================================= */

function formatNumber(value){
    if(!Number.isFinite(value)) return "∞";
    const abs=Math.abs(value);
    if(abs<1000) return Math.floor(value).toLocaleString();

    const suffix=["","K","M","B","T","Qa","Qi","Sx","Sp","Oc","No","Dc"];
    const tier=Math.floor(Math.log10(abs)/3);
    
    if(tier>=suffix.length) return value.toExponential(3);
    return (value/Math.pow(1000,tier)).toFixed(2)+suffix[tier];
}

/* =========================================================
   REALM / PRESTIGE HELPERS
========================================================= */

function currentRealm(){
    if(game.unknownUnlocked){
        return { name:"??? — Unknown Realm", unknown:true };
    }
    return normalRealms[game.realm];
}

function prestigeMultiplier(){
    return Math.pow(1.25, game.reincarnations);
}

function masteryMultiplier(){
    return 1 + (game.weaponMastery * 0.05);
}

function unknownPower(){
    if(!game.unknownUnlocked) return 1;
    const cultivation = Math.max(1, game.cultivation);
    return 180 * Math.max(1, Math.log10(cultivation+10)) * prestigeMultiplier();
}

function realmMultiplier(){
    if(game.unknownUnlocked) return unknownPower();
    return normalRealms[game.realm].mult * prestigeMultiplier();
}


/* =========================================================
   EFFECTIVE STATS
========================================================= */

function calculateBaseAtk(){
    return game.baseAtk * realmMultiplier() * masteryMultiplier();
}

function calculateWeaponAtk(){
    if(game.equippedWeapon === null) return 0;
    
    const w = weaponsData[game.equippedWeapon];
    if(w.unknown){
        // Scales mathematically with Unknown Cultivation
        return Math.floor(Math.pow(Math.max(1, game.cultivation), 0.45) * 150 * prestigeMultiplier() * masteryMultiplier());
    } else {
        return w.atk * realmMultiplier() * masteryMultiplier();
    }
}

function effectiveAtk(){
    let total = calculateBaseAtk() + calculateWeaponAtk();
    if(game.equipment.includes("Heavenly Sword Fragment")){
        total += 30;
    }
    return total;
}

function effectiveDef(){
    let value = game.baseDef * realmMultiplier();
    if(game.techniques.includes(1)) value *= 1.25;
    if(game.equipment.includes("Iron Guardian Talisman")) value += 20;
    return value;
}

function effectiveMaxHp(){
    let value = game.maxHp * realmMultiplier();
    if(game.equipment.includes("Phoenix Feather")) value *= 1.10;
    return value;
}


/* =========================================================
   CULTIVATION RATE
========================================================= */

function cultivationRate(){
    let rate = 0.8 * realmMultiplier();
    if(game.techniques.includes(0)) rate *= 1.20;
    if(game.equipment.includes("Jade Spirit Ring")) rate *= 1.15;
    return rate;
}

    return Math.floor(100 * Math.pow(game.level, 1.65));
}

/* =========================================================
   UPGRADE COSTS
========================================================= */

function upgradeCost(stat){
    const base = stat==="atk" ? 50 : 60;
    const value = stat==="atk" ? game.baseAtk : game.baseDef;
    return Math.floor(base * Math.pow(value/(stat==="atk"?10:5), 1.35));
}

   NOTIFICATIONS
========================================================= */

function notify(text){
    const notice = document.getElementById("notice");
    notice.textContent = text;
    notice.classList.add("show");
    setTimeout(() => { notice.classList.remove("show"); }, 1800);
}

function log(text, cls=""){
    const box = document.getElementById("log");
    const div = document.createElement("div");
    div.className = cls;
    div.textContent = text;
    box.appendChild(div);
    box.scrollTop = box.scrollHeight;
}


/* =========================================================
   RANDOM HELPERS
========================================================= */

function rand(min,max){
    return Math.floor(Math.random()*(max-min+1))+min;
}

function chance(probability){
    return Math.random()<probability;
}

/* =========================================================
   SPIRIT STONE -> CULTIVATION CONVERSION
========================================================= */

function convertStones(tier){
    let cost = 0;
    let gained = 0;

    if(tier === 1){
        cost = 100;
        gained = 100;
    } else if(tier === 2){
        cost = 1000;
        gained = 1100;
    } else if(tier === 3){
        cost = 10000;
        gained = 12500;
    } else if(tier === 4 && game.unknownUnlocked){
        cost = Math.floor(game.stones * 0.5);
        if(cost < 1){
            notify("Not enough Spirit Stones to scale conversion.");
            return;
        }
        gained = Math.floor(cost * 1.5);
    } else {
        return;
    }

    if(game.stones < cost){
        notify("Not enough Spirit Stones.");
        return;
    }

    game.stones -= cost;
    game.cultivation += gained;

    notify(`Refined ${formatNumber(cost)} 💠`);
    log(`☯ Refined ${formatNumber(cost)} Spirit Stones into ${formatNumber(gained)} Cultivation.`, "good");

    update();
}

/* =========================================================
   BREAKTHROUGH
========================================================= */

function breakthrough(){
    if(game.unknownUnlocked){
        notify("The Unknown Realm has no finite breakthrough threshold.");
        return;
    }

    const req = normalRealms[game.realm].req;

    /* Final normal realm */
    if(game.realm === normalRealms.length-1){
        if(game.cultivation < req){
            notify("Immortal Ascension is not completely maxed.");
            return;
        }

        if(game.reincarnations < 10){
            notify("You need " + (10-game.reincarnations) + " more reincarnation(s).");
            return;
        }

        game.unknownUnlocked = true;
        game.cultivation = 1;
        game.maxHp *= 10;
        game.hp = effectiveMaxHp();
        game.baseAtk *= 5;
        game.baseDef *= 5;

        // Auto unlock the Unknown Weapon
        if(!game.weapons.includes(9)){
            game.weapons.push(9);
            game.weaponMastery = Math.max(game.weaponMastery, 9);
            game.equippedWeapon = 9;
            log("🌌 The Heavens shatter, and the ??? — Unknown Weapon manifests in your grasp.", "mythic");
        }

        log("☯ THE HEAVENS HAVE BEEN SURPASSED.", "good");
        log("You have entered the ??? — Unknown Realm.", "good");
        notify("UNKNOWN REALM UNLOCKED!");
        update();
        return;
    }

    if(game.cultivation < req){
        notify("Insufficient cultivation.");
        return;
    }

    const successChance = Math.min(0.92, 0.45 + game.realm*0.045);

    if(chance(successChance)){
        game.realm++;
        game.cultivation = 0;
        game.baseAtk *= 1.18;
        game.baseDef *= 1.18;
        game.maxHp *= 1.15;
        game.hp = effectiveMaxHp();

        log("☯ Breakthrough successful: " + normalRealms[game.realm].name, "good");
        notify("Breakthrough successful!");
    }else{
        game.cultivation *= 0.72;
        game.hp = Math.max(1, game.hp*0.65);
        log("☯ Breakthrough failed. Your cultivation slips.", "bad");
        notify("Breakthrough failed.");
    }
    update();
}


/* =========================================================
   REINCARNATION
========================================================= */

function reincarnate(){
    if(game.unknownUnlocked){
        notify("You have already surpassed reincarnation.");
        return;
    }

    const finalRealm = normalRealms.length-1;

    if(game.realm !== finalRealm){
        notify("You must reach Immortal Ascension first.");
        log("☯ Reincarnation unavailable: current realm is " + normalRealms[game.realm].name + ".", "bad");
        return;
    }

    if(game.cultivation < normalRealms[finalRealm].req){
        const remaining = normalRealms[finalRealm].req - game.cultivation;
        notify("Immortal Ascension is not maxed.");
        log("☯ You need " + formatNumber(remaining) + " more cultivation to reincarnate.", "bad");
        return;
    }

    const nextCount = game.reincarnations + 1;

    const confirmed = confirm(
        "☯ REINCARNATE?\n\n" +
        "You will return to Mortal.\n\n" +
        "Your normal cultivation, realm, XP, Spirit Stones, techniques, normal weapons, and equipment will reset.\n\n" +
        "Your permanent reincarnation count, prestige power, Weapon Mastery, and Soulbound Mythic weapons will remain.\n\n" +
        "Reincarnation: " + game.reincarnations + " → " + nextCount + "\n\n" +
        "Permanent bonus after reincarnation: " + formatMultiplier(Math.pow(1.25, nextCount)) + "\n\n" +
        (nextCount>=10 ? "After reaching 10 reincarnations, you can eventually break beyond Immortal Ascension." : "")
    );

    if(!confirmed) return;

    /* Preserve permanent progression */
    game.reincarnations = nextCount;
    
    // Preserve soulbound weapons (index 8 and 9)
    const keptWeapons = game.weapons.filter(idx => weaponsData[idx].soulbound === true);
    const keptEquipped = (game.equippedWeapon !== null && weaponsData[game.equippedWeapon].soulbound === true) ? game.equippedWeapon : null;
    const keptMastery = game.weaponMastery;

    /* Reset temporary progression */
    game.realm = 0;
    game.level = 1;
    game.cultivation = 0;
    game.xp = 0;
    game.stones = 100;
    game.baseAtk = 10;
    game.baseDef = 5;
    game.hp = 100;
    game.maxHp = 100;
    game.techniques = [0];
    game.equipment = [];
    game.weapons = keptWeapons;
    game.equippedWeapon = keptEquipped;
    game.weaponMastery = keptMastery;
    
    game.combat = false;
    game.enemy = null;
    game.defending = false;
    game.lastSave = Date.now();

    log("☯ Reincarnation " + game.reincarnations + " completed.", "good");
    log("Your memories fade, but your eternal foundation remains.", "good");
    notify("Reincarnation successful!");

    update();
    saveGame();
}

function formatMultiplier(value){
    return "×" + value.toFixed(3);
}

    if(game.stones < cost){
        notify("Not enough Spirit Stones.");
        return;
    }
    game.stones -= cost;

    if(stat==="atk"){
        game.baseAtk++;
    }else{
        game.baseDef++;

    notify("Foundation strengthened.");
    update();
}


/* =========================================================
   EXPLORATION
========================================================= */

function renderAreas(){
    const box = document.getElementById("areas");
    box.innerHTML = "";

    areas.forEach((area,index)=>{
        if(area.unknown && !game.unknownUnlocked) return;

        let locked = false;
        if(!area.unknown) locked = game.realm < area.req;

        if(area.unknown){
            box.innerHTML+=`
                <div class="card">

                    ${locked ? "🔒 Locked" : "🌌 Enter"}
                </button>
            </div>
        `;
    });

function explore(index){
    const area = areas[index];
    if(game.realm < area.req) return;

    if(chance(0.25)){
        randomEvent(1);
        return;
    }

    const enemyIndex = area.enemies[rand(0, area.enemies.length-1)];
    startCombat(enemyIndex);
    showTab("combat");
}

function exploreUnknown(){
    if(!game.unknownUnlocked){
        notify("The Unknown Realm is locked.");
        return;
    }
    startUnknownCombat();
    showTab("combat");
}

}


/* =========================================================
   RANDOM EVENTS
========================================================= */

function randomEvent(force = 0.1){
    if(!chance(force)) return;

    const events = [
        ()=>{
            const gain = rand(10,40) * (game.realm+1) * prestigeMultiplier();
            game.stones += gain;
            log("🌿 You discover a hidden Spirit Stone vein.", "good");
            notify("+" + formatNumber(gain) + " Spirit Stones!");
        },
        ()=>{
            const gain = rand(5,15) * (game.realm+1) * prestigeMultiplier();
            game.cultivation += gain;
            log("☯ A spiritual spring nourishes your meridians.", "good");
            notify("+" + formatNumber(gain) + " Cultivation");
        },
        ()=>{
            game.hp = Math.min(effectiveMaxHp(), game.hp + effectiveMaxHp()*0.25);
            log("🌸 You discover a rare healing herb.", "good");
        },
        ()=>{
            game.stones = Math.max(0, game.stones - rand(5,20));
            log("🐍 A venomous creature steals some Spirit Stones.", "bad");
        }
    ];

    events[rand(0, events.length-1)]();
    update();
}


/* =========================================================
   COMBAT
========================================================= */

function startCombat(index){
    const enemy = enemies[index];
    const scale = 1 + Math.max(0, game.realm - enemy.realm) * 0.12;

    game.enemy = {
        ...enemy,
        maxHp: enemy.hp * scale,
        hp: enemy.hp * scale,
        atk: enemy.atk * scale,
        def: enemy.def * scale
    };

    game.combat = true;
    game.defending = false;

    log("⚔ You encounter a " + enemy.name + "!", "bad");
    renderEnemy();
    update();
}

function damage(attacker, defender){
    return Math.max(1, attacker - defender*0.55);
}

/* =========================================================
   PLAYER ATTACK
========================================================= */

function playerAttack(){
    if(!game.combat) return;

    let dmg = damage(effectiveAtk(), game.enemy.def);
    const crit = chance(0.12);
    if(crit) dmg *= 2;
    dmg = Math.floor(dmg);

    game.enemy.hp -= dmg;
    log("⚔ You strike for " + formatNumber(dmg) + (crit ? " CRITICAL DAMAGE!" : " damage."), "good");

    if(game.enemy.hp <= 0){
        victory();
        return;
    }

    enemyTurn();
    update();
}


/* =========================================================
   DEFEND
========================================================= */

function defend(){
    if(!game.combat) return;
    game.defending = true;
    log("🛡 You brace yourself.", "good");
    enemyTurn();
}


/* =========================================================
   TECHNIQUE COMBAT
========================================================= */

function useTechnique(index){
    if(!game.combat) return;

    const technique = techniques[index];
    if(!game.techniques.includes(index) || !technique.skill) return;

    let dmg = damage(effectiveAtk() * technique.skill, game.enemy.def);
    if(chance(0.15)) dmg *= 2;
    dmg = Math.floor(dmg);

    game.enemy.hp -= dmg;
    log("📜 " + technique.name + " deals " + formatNumber(dmg) + " damage!", "good");

    if(technique.burn && game.enemy.hp > 0){
        const burn = effectiveAtk() * 0.5;
        game.enemy.hp -= burn;
        log("🔥 Burning energy deals " + formatNumber(burn) + " additional damage.", "good");
    }

    if(game.enemy.hp <= 0){
        victory();
        return;
    }

    enemyTurn();
    update();
}


/* =========================================================
   ENEMY TURN
========================================================= */

function enemyTurn(){
    if(!game.combat) return;

    let dmg = damage(game.enemy.atk, effectiveDef());
    
    if(game.defending){
        dmg *= 0.35;
        game.defending = false;
    }

    if(chance(0.08)){
        dmg *= 2;
        log("☠ " + game.enemy.name + " lands a CRITICAL HIT!", "bad");
    }

    dmg = Math.floor(dmg);
    game.hp -= dmg;
    log("☠ " + game.enemy.name + " hits you for " + formatNumber(dmg) + ".", "bad");

    if(game.hp <= 0){
        game.hp = 1;
        game.stones *= 0.85;
        game.cultivation *= 0.8;
        game.combat = false;
        game.enemy = null;
        log("☯ You barely escape with your life.", "bad");
        notify("Defeated — resources lost.");
    }

    renderEnemy();
}


/* =========================================================
   VICTORY
========================================================= */

function victory(){
    const enemy = game.enemy;
    game.combat = false;
    game.enemy = null;

    const multiplier = enemy.unknown ? prestigeMultiplier() : 1 + game.realm * 0.08;
    const stones = enemy.stones * multiplier;
    const xp = enemy.xp * multiplier;

    game.stones += stones;
    game.xp += xp;

    log("🏆 Victory over " + enemy.name + "!", "good");
    log("+" + formatNumber(stones) + " Spirit Stones.", "gold");
    log("+" + formatNumber(xp) + " experience.", "good");

    checkLevel();

    if(enemy.unknown){
        const cultivationGain = xp * Math.max(1, Math.log10(game.cultivation + 10));
        game.cultivation += cultivationGain;
        log("☯ You absorb residual Unknown energy.", "good");
    }else if(chance(0.18)){
        const cultivationGain = xp * (game.realm + 1);
        game.cultivation += cultivationGain;
        log("☯ You absorb residual spiritual energy.", "good");
    }

    if(chance(0.07)) findArtifact();

    update();
}


/* =========================================================
   LEVELING
========================================================= */

function checkLevel(){
    while(game.xp >= xpNeeded()){
        game.xp -= xpNeeded();
        game.level++;
        game.baseAtk += 2;
        game.baseDef++;
        game.maxHp += 15;
        game.hp = effectiveMaxHp();
        log("⭐ You reached level " + game.level + "!", "good");
    }
}


/* =========================================================
   TECHNIQUES UI
========================================================= */

function renderTechniques(){
    const box = document.getElementById("techList");
    box.innerHTML = "";

    techniques.forEach((technique,index)=>{

        const locked = !game.unknownUnlocked && game.realm < technique.req;

        let button;

            button = `<button disabled>✓ Learned</button>`;
        }else if(locked){
            button = `<button disabled>🔒 Requires ${normalRealms[technique.req].name}</button>`;
        }else{
            button = `<button onclick="learnTechnique(${index})">Learn — 💠 ${formatNumber(technique.cost)}</button>`;
        }

        box.innerHTML+=`
            <div class="tech ${locked&&!owned ? "locked" : ""}">
                <strong>${technique.name}</strong>
                <span class="small">— ${technique.quality}</span>
                <p class="small">${technique.desc}</p>
                ${button}
            </div>
        `;
    });

    renderCombatTechniques();
}

function renderCombatTechniques(){
    const box = document.getElementById("combatTechniques");
    box.innerHTML = "";

    if(!game.combat) return;

    game.techniques.forEach(index=>{
        const technique = techniques[index];
        if(!technique.skill) return;

        const button = document.createElement("button");
        button.className = "full";
        button.textContent = "📜 " + technique.name;
        button.onclick = () => useTechnique(index);
        box.appendChild(button);
    });
}

function learnTechnique(index){
    const technique = techniques[index];
    if(game.techniques.includes(index)) return;
    if(!game.unknownUnlocked && game.realm < technique.req){
        notify("Your realm is too low."); return;
    }
    if(game.stones < technique.cost){
        notify("Not enough Spirit Stones."); return;
    }

    game.stones -= technique.cost;
    game.techniques.push(index);
    log("📜 You learned " + technique.name + "!", "good");
    notify("Technique learned.");
    update();
}

/* =========================================================
   WEAPONS UI & LOGIC
========================================================= */

function renderWeapons(){
    const box = document.getElementById("weaponList");
    box.innerHTML = "";

    weaponsData.forEach((w, index) => {
        if(w.unknown && !game.unknownUnlocked) return; // Hide Unknown Weapon entirely until unlocked

        const owned = game.weapons.includes(index);
        const equipped = (game.equippedWeapon === index);
        const locked = !game.unknownUnlocked && game.realm < w.req;
        
        let actions = "";
        
        if(owned){
            if(equipped){
                actions = `<button class="unequip-btn" onclick="unequipWeapon()">Unequip</button>`;
            } else {
                actions = `<button class="equip-btn" onclick="equipWeapon(${index})">Equip</button>`;
            }
        } else if(locked){
            actions = `<button disabled>🔒 Requires ${normalRealms[w.req].name}</button>`;
        } else {
            actions = `<button onclick="buyWeapon(${index})">Buy — 💠 ${formatNumber(w.cost)}</button>`;
        }

        // Show a dynamic text for Unknown Weapon Atk instead of 0
        let displayAtk = w.atk;
        if(w.unknown) displayAtk = "Scaling";

        box.innerHTML+=`
            <div class="weapon-card ${locked&&!owned ? "locked" : ""}" style="${equipped ? 'border-color:#5c7841;' : ''}">
                <div class="stat" style="margin-top:0;">
                    <strong style="${w.soulbound ? 'color:#c471ed;' : 'color:#d5b963'}">${w.name} ${equipped ? " (Equipped)" : ""}</strong>
                    <span class="small">${w.quality}</span>
                </div>
                <div class="stat small" style="margin:2px 0 6px 0;">
                    <span>⚔ Base Atk: ${displayAtk}</span>
                </div>
                <p class="small" style="margin-bottom:10px;">${w.desc}</p>
                ${actions}
            </div>
        `;
    });
}

function buyWeapon(index){
    const w = weaponsData[index];
    if(game.weapons.includes(index)) return;
    
    if(!game.unknownUnlocked && game.realm < w.req){
        notify("Your realm is too low."); return;
    }
    if(game.stones < w.cost){
        notify("Not enough Spirit Stones."); return;
    }

    game.stones -= w.cost;
    game.weapons.push(index);
    
    // Update Mastery
    if(index > game.weaponMastery){
        game.weaponMastery = index;
    }

    log("⚔ You acquired the " + w.name + "!", "good");
    notify("Weapon purchased.");
    update();
}

function equipWeapon(index){
    if(!game.weapons.includes(index)) return;
    game.equippedWeapon = index;
    notify("Weapon equipped.");
    update();
}

function unequipWeapon(){
    game.equippedWeapon = null;
    notify("Weapon unequipped.");
    update();
}

function findArtifact(){
    const artifact = artifacts[rand(0, artifacts.length-1)];

    if(game.equipment.includes(artifact.name)){
        game.stones += 100;
        log("You found another " + artifact.name + " and sold it for 100 Spirit Stones.", "gold");
        return;
    }

    game.equipment.push(artifact.name);
    log("✨ Artifact discovered: " + artifact.name + " — " + artifact.desc, "good");
    notify("Rare artifact found!");
}

function renderInventory(){
    const box = document.getElementById("inventoryList");

    if(game.equipment.length === 0){
        box.innerHTML = `<div class="card small">Your inventory is empty.</div>`;
        return;
    }

    box.innerHTML = "";
    game.equipment.forEach(name=>{
        const artifact = artifacts.find(item=>item.name===name);
        box.innerHTML+=`
            <div class="card">
                <strong>✨ ${artifact.name}</strong>
                <p class="small">${artifact.desc}</p>
            </div>
        `;
    });
}

            </div>

        </div>
    `;
}


/* =========================================================
   HEAL
========================================================= */

function heal(){
    const max = effectiveMaxHp();
    const missing = Math.max(0, max-game.hp);

    if(missing <= 0){
        notify("Your body is already restored.");
        return;
    }

    const cost = Math.max(5, Math.floor(missing*0.15));

    if(game.stones < cost){
        notify("Not enough Spirit Stones.");
        return;
    }

    game.stones -= cost;
    game.hp = max;
    notify("Body restored.");
    update();
}


/* =========================================================
   SAVE / LOAD
========================================================= */

function saveGame(){
    game.lastSave = Date.now();
    localStorage.setItem("pathOfTheImmortalSave", JSON.stringify(game));
    notify("Game saved.");
}

function loadGame(){
    const raw = localStorage.getItem("pathOfTheImmortalSave");
    if(!raw) return;

    try{
        const saved = JSON.parse(raw);
        game = Object.assign(structuredClone(defaultGame), saved);
        
        // Clean up legacy property to respect "No Qi" rule unconditionally
        delete game.qi;

        processOffline();
    }catch(error){
        console.error("Save load failed:", error);
    }
}


/* =========================================================
   OFFLINE CULTIVATION
========================================================= */

function processOffline(){
    const now = Date.now();
    const seconds = Math.min((now - game.lastSave)/1000, 60*60*24);

    if(seconds > 5){
        const gain = cultivationRate() * seconds;
        game.cultivation += gain;
        notify("Offline cultivation: +" + formatNumber(gain) + " Cultivation");
    }
    game.lastSave = now;
}
