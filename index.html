/* =========================================================

NUMBER FORMATTING

========================================================= */


function formatNumber(value){

if(!Number.isFinite(value)) return "∞";

const abs=Math.abs(value);

if(abs<1000) return Math.floor(value).toLocaleString();


const suffix=["","K","M","B","T","Qa","Qi","Sx","Sp","Oc","No","Dc"];

const tier=Math.floor(Math.log10(abs)/3);

if(tier>=suffix.length) return value.toExponential(3);

return (value/Math.pow(1000,tier)).toFixed(2)+suffix[tier];

}

function prestigeMultiplier(){

return Math.pow(1.25, game.reincarnations);

}


function masteryMultiplier(){

return 1 + (game.weaponMastery * 0.05);

}


function unknownPower(){

if(!game.unknownUnlocked) return 1;

const cultivation = Math.max(1, game.cultivation);

return 180 * Math.max(1, Math.log10(cultivation+10)) * prestigeMultiplier();

}


function realmMultiplier(){

if(game.unknownUnlocked) return unknownPower();

return normalRealms[game.realm].mult * prestigeMultiplier();

}



/* =========================================================

EFFECTIVE STATS

========================================================= */


function calculateBaseAtk(){

return game.baseAtk * realmMultiplier() * masteryMultiplier();

}


function calculateWeaponAtk(){

if(game.equippedWeapon === null) return 0;

const w = weaponsData[game.equippedWeapon];

if(w.unknown){

// Scales mathematically with Unknown Cultivation

return Math.floor(Math.pow(Math.max(1, game.cultivation), 0.45) * 150 * prestigeMultiplier() * masteryMultiplier());

} else {

return w.atk * realmMultiplier() * masteryMultiplier();

}

}

return total;

}

function convertStones(tier){

let cost = 0;

let gained = 0;


if(tier === 1){

cost = 100;

gained = 100;

} else if(tier === 2){

cost = 1000;

gained = 1100;

} else if(tier === 3){

cost = 10000;

gained = 12500;

} else if(tier === 4 && game.unknownUnlocked){

cost = Math.floor(game.stones * 0.5);

if(cost < 1){

notify("Not enough Spirit Stones to scale conversion.");

return;

}

gained = Math.floor(cost * 1.5);

} else {

return;

}


if(game.stones < cost){

notify("Not enough Spirit Stones.");

return;

}


game.stones -= cost;

game.cultivation += gained;


notify(`Refined ${formatNumber(cost)} 💠`);

log(`☯ Refined ${formatNumber(cost)} Spirit Stones into ${formatNumber(gained)} Cultivation.`, "good");


update();

}

/* =========================================================

REINCARNATION

========================================================= */


function reincarnate(){

if(game.unknownUnlocked){

notify("You have already surpassed reincarnation.");

return;

}


const finalRealm = normalRealms.length-1;


if(game.realm !== finalRealm){

notify("You must reach Immortal Ascension first.");

log("☯ Reincarnation unavailable: current realm is " + normalRealms[game.realm].name + ".", "bad");

return;

}


if(game.cultivation < normalRealms[finalRealm].req){

const remaining = normalRealms[finalRealm].req - game.cultivation;

notify("Immortal Ascension is not maxed.");

log("☯ You need " + formatNumber(remaining) + " more cultivation to reincarnate.", "bad");

return;

}


const nextCount = game.reincarnations + 1;


const confirmed = confirm(

"☯ REINCARNATE?\n\n" +

"You will return to Mortal.\n\n" +

"Your normal cultivation, realm, XP, Spirit Stones, techniques, normal weapons, and equipment will reset.\n\n" +

"Your permanent reincarnation count, prestige power, Weapon Mastery, and Soulbound Mythic weapons will remain.\n\n" +

"Reincarnation: " + game.reincarnations + " → " + nextCount + "\n\n" +

"Permanent bonus after reincarnation: " + formatMultiplier(Math.pow(1.25, nextCount)) + "\n\n" +

(nextCount>=10 ? "After reaching 10 reincarnations, you can eventually break beyond Immortal Ascension." : "")

);


if(!confirmed) return;


/* Preserve permanent progression */

game.reincarnations = nextCount;

// Preserve soulbound weapons (index 8 and 9)

const keptWeapons = game.weapons.filter(idx => weaponsData[idx].soulbound === true);

const keptEquipped = (game.equippedWeapon !== null && weaponsData[game.equippedWeapon].soulbound === true) ? game.equippedWeapon : null;

const keptMastery = game.weaponMastery;


/* Reset temporary progression */

game.realm = 0;

game.level = 1;

game.cultivation = 0;

game.xp = 0;

game.stones = 100;

game.baseAtk = 10;

game.baseDef = 5;

game.techniques = [0];

game.equippedWeapon = keptEquipped;

game.weaponMastery = keptMastery;

game.combat = false;

game.enemy = null;

game.defending = false;

game.lastSave = Date.now();


log("☯ Reincarnation " + game.reincarnations + " completed.", "good");

log("Your memories fade, but your eternal foundation remains.", "good");

notify("Reincarnation successful!");


update();

saveGame();

}

========================================================= */


function useTechnique(index){

if(!game.combat) return;


const technique = techniques[index];

if(!game.techniques.includes(index) || !technique.skill) return;


let dmg = damage(effectiveAtk() * technique.skill, game.enemy.def);

if(chance(0.15)) dmg *= 2;

dmg = Math.floor(dmg);


game.enemy.hp -= dmg;

log("📜 " + technique.name + " deals " + formatNumber(dmg) + " damage!", "good");


if(technique.burn && game.enemy.hp > 0){

const burn = effectiveAtk() * 0.5;

game.enemy.hp -= burn;

log("🔥 Burning energy deals " + formatNumber(burn) + " additional damage.", "good");

}


if(game.enemy.hp <= 0){

victory();

return;

}


enemyTurn();

update();

}



/* =========================================================

ENEMY TURN

========================================================= */


function enemyTurn(){

if(!game.combat) return;


let dmg = damage(game.enemy.atk, effectiveDef());

if(game.defending){

dmg *= 0.35;

game.defending = false;

}


if(chance(0.08)){

dmg *= 2;

log("☠ " + game.enemy.name + " lands a CRITICAL HIT!", "bad");

}


dmg = Math.floor(dmg);

game.hp -= dmg;

log("☠ " + game.enemy.name + " hits you for " + formatNumber(dmg) + ".", "bad");


if(game.hp <= 0){

game.hp = 1;

game.stones *= 0.85;

game.cultivation *= 0.8;

game.combat = false;

game.enemy = null;

log("☯ You barely escape with your life.", "bad");

notify("Defeated — resources lost.");

}


renderEnemy();

}



/* =========================================================

VICTORY

========================================================= */


function victory(){

const enemy = game.enemy;

game.combat = false;

game.enemy = null;


const multiplier = enemy.unknown ? prestigeMultiplier() : 1 + game.realm * 0.08;

const stones = enemy.stones * multiplier;

const xp = enemy.xp * multiplier;


game.stones += stones;

game.xp += xp;


log("🏆 Victory over " + enemy.name + "!", "good");

log("+" + formatNumber(stones) + " Spirit Stones.", "gold");

log("+" + formatNumber(xp) + " experience.", "good");


checkLevel();


if(enemy.unknown){

const cultivationGain = xp * Math.max(1, Math.log10(game.cultivation + 10));

game.cultivation += cultivationGain;

log("☯ You absorb residual Unknown energy.", "good");

}else if(chance(0.18)){

const cultivationGain = xp * (game.realm + 1);

game.cultivation += cultivationGain;

log("☯ You absorb residual spiritual energy.", "good");

}


if(chance(0.07)) findArtifact();


update();

}

/* =========================================================

HEAL

========================================================= */


function heal(){

const max = effectiveMaxHp();

const missing = Math.max(0, max-game.hp);


if(missing <= 0){

notify("Your body is already restored.");

return;

}


const cost = Math.max(5, Math.floor(missing*0.15));


if(game.stones < cost){

notify("Not enough Spirit Stones.");

return;

}


game.stones -= cost;

game.hp = max;

notify("Body restored.");

update();

}



/* =========================================================

SAVE / LOAD

========================================================= */


function saveGame(){

game.lastSave = Date.now();

localStorage.setItem("pathOfTheImmortalSave", JSON.stringify(game));

notify("Game saved.");

}


function loadGame(){

const raw = localStorage.getItem("pathOfTheImmortalSave");

if(!raw) return;


try{

const saved = JSON.parse(raw);

game = Object.assign(structuredClone(defaultGame), saved);

// Clean up legacy property to respect "No Qi" rule unconditionally

delete game.qi;


processOffline();

}catch(error){

console.error("Save load failed:", error);

}

}



/* =========================================================

OFFLINE CULTIVATION

========================================================= */


function processOffline(){

const now = Date.now();

const seconds = Math.min((now - game.lastSave)/1000, 60*60*24);


if(seconds > 5){

const gain = cultivationRate() * seconds;

game.cultivation += gain;

notify("Offline cultivation: +" + formatNumber(gain) + " Cultivation");

}

game.lastSave = now;

}
